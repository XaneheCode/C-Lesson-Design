# 文具店销售管理系统 — 课程设计报告核心代码

## 一、项目概述

本系统是一个基于 C 语言的文具店销售管理系统，采用 **HTTP 服务器 + RESTful API** 架构。后端使用 Winsock2 实现 TCP 服务器，前端为浏览器 HTML 页面。数据持久化采用管道符分隔的文本文件（类 CSV 格式）。

**核心模块**: 数据层 (data) → 商品管理 (product) → 销售管理 (sale) → 进货管理 (purchase) → 统计分析 (stats) → HTTP 路由 (server) → 工具函数 (utils)

```c
/* ===== 文件格式示例 =====
 * products.txt:
 *   id|name|category|manufacturer|model|stock|price
 *   P001|晨光中性笔|笔|晨光文具|K35|150|2.50
 */

/* 解析一行商品文本 */
static int parse_product(const char *line, Product *p) {
    char tmp[512];
    strncpy(tmp, line, sizeof(tmp) - 1);
    tmp[sizeof(tmp) - 1] = '\0';

    char *tok = strtok(tmp, "|");
    if (!tok) return 0;
    strncpy(p->id, tok, STR_LEN - 1);

    tok = strtok(NULL, "|");
    if (!tok) return 0;
    strncpy(p->name, tok, STR_LEN - 1);

    tok = strtok(NULL, "|");
    if (!tok) return 0;
    strncpy(p->category, tok, STR_LEN - 1);

    tok = strtok(NULL, "|");
    if (!tok) return 0;
    strncpy(p->manufacturer, tok, STR_LEN - 1);

    tok = strtok(NULL, "|");
    if (!tok) return 0;
    strncpy(p->model, tok, STR_LEN - 1);

    tok = strtok(NULL, "|");
    if (!tok) return 0;
    p->stock = atoi(tok);

    tok = strtok(NULL, "|\n\r");
    if (!tok) return 0;
    p->price = atof(tok);

    return 1;
}

/* 加载商品数据 */
void load_products(void) {
    FILE *fp = fopen("data/products.txt", "r");
    if (!fp) { product_count = 0; return; }

    char line[512];
    product_count = 0;

    /* 跳过第一行表头 */
    if (fgets(line, sizeof(line), fp) == NULL) { fclose(fp); return; }

    while (fgets(line, sizeof(line), fp) && product_count < MAX_PRODUCTS) {
        if (strlen(line) < 5) continue;
        if (parse_product(line, &products[product_count])) {
            product_count++;
        }
    }
    fclose(fp);
}

/* 保存商品数据 */
void save_products(void) {
    FILE *fp = fopen("data/products.txt", "w");
    if (!fp) return;

    fprintf(fp, "id|name|category|manufacturer|model|stock|price\n");
    for (int i = 0; i < product_count; i++) {
        fprintf(fp, "%s|%s|%s|%s|%s|%d|%.2f\n",
            products[i].id, products[i].name, products[i].category,
            products[i].manufacturer, products[i].model,
            products[i].stock, products[i].price);
    }
    fclose(fp);
}

/* 自动递增生成下一个商品编号 */
void next_product_id(char *buf) {
    int max_num = 0;
    for (int i = 0; i < product_count; i++) {
        if (strncmp(products[i].id, "P", 1) == 0) {
            int num = atoi(products[i].id + 1);
            if (num > max_num) max_num = num;
        }
    }
    sprintf(buf, "P%03d", max_num + 1);
}

/* 按编号查找商品 */
int find_product_by_id(const char *id) {
    for (int i = 0; i < product_count; i++) {
        if (strcmp(products[i].id, id) == 0) {
            return i;
        }
    }
    return -1;
}
```

```c
/* JsonBuilder — JSON 字符串构建器
 * 由于 C 语言没有内置 JSON 库，我们自己实现了一个轻量级构建器 */
typedef struct {
    char *buf;   /* 输出缓冲区 */
    int   len;   /* 当前写入位置 */
    int   cap;   /* 缓冲区总容量 */
    int   first; /* 是否为第一个字段（控制逗号分隔符） */
} JsonBuilder;

void json_init(JsonBuilder *jb, char *buf, int cap) {
    jb->buf = buf;
    jb->cap = cap;
    jb->len = 0;
    jb->first = 1;
    buf[0] = '\0';
}

void json_add_str(JsonBuilder *jb, const char *key, const char *val) {
    char escaped[512];
    json_escape(val, escaped, sizeof(escaped));  /* 先转义特殊字符 */

    char tmp[1024];
    if (!jb->first) json_append_raw(jb, ",");    /* 非首个字段加逗号 */
    sprintf(tmp, "\"%s\":\"%s\"", key, escaped);
    json_append_raw(jb, tmp);
    jb->first = 0;
}

void json_add_int(JsonBuilder *jb, const char *key, int val) {
    char tmp[256];
    if (!jb->first) json_append_raw(jb, ",");
    sprintf(tmp, "\"%s\":%d", key, val);
    json_append_raw(jb, tmp);
    jb->first = 0;
}

void json_add_dbl(JsonBuilder *jb, const char *key, double val) {
    char tmp[256];
    if (!jb->first) json_append_raw(jb, ",");
    sprintf(tmp, "\"%s\":%.2f", key, val);
    json_append_raw(jb, tmp);
    jb->first = 0;
}

void json_begin_array(JsonBuilder *jb, const char *key) {
    char tmp[256];
    if (!jb->first) json_append_raw(jb, ",");
    sprintf(tmp, "\"%s\":[", key);
    json_append_raw(jb, tmp);
    jb->first = 1;  /* 数组内首个元素不加逗号 */
}

void json_begin_obj(JsonBuilder *jb) {
    json_append_raw(jb, "{");
    jb->first = 1;
}

void json_end_obj(JsonBuilder *jb) {
    json_append_raw(jb, "}");
    jb->first = 0;
}

/* JSON 字符串转义: " → \"   \ → \\   换行 → \n  等 */
void json_escape(const char *src, char *dst, int dst_size) {
    int j = 0;
    for (int i = 0; src[i] && j < dst_size - 2; i++) {
        switch (src[i]) {
            case '"':  dst[j++] = '\\'; dst[j++] = '"';  break;
            case '\\': dst[j++] = '\\'; dst[j++] = '\\'; break;
            case '\n': dst[j++] = '\\'; dst[j++] = 'n';  break;
            case '\r': dst[j++] = '\\'; dst[j++] = 'r';  break;
            case '\t': dst[j++] = '\\'; dst[j++] = 't';  break;
            default:   dst[j++] = src[i]; break;
        }
    }
    dst[j] = '\0';
}

/* URL 百分号解码: %E7%AC%94 → 笔 */
void url_decode(char *src) {
    char *dst = src;
    while (*src) {
        if (*src == '%' && src[1] && src[2]) {
            char hex[3] = {src[1], src[2], 0};
            *dst++ = (char)strtol(hex, NULL, 16);
            src += 3;
        } else if (*src == '+') {
            *dst++ = ' ';
            src++;
        } else {
            *dst++ = *src++;
        }
    }
    *dst = '\0';
}
```

```c
/* 从 JSON 字符串中提取字符串字段值 */
int json_get_str(const char *json, const char *key, char *out, int out_len) {
    /* 尝试匹配带引号的值 "key":"value" */
    char search[128];
    sprintf(search, "\"%s\":\"", key);
    const char *p = strstr(json, search);

    if (!p) {
        /* 尝试无引号模式 "key": (用于数值) */
        sprintf(search, "\"%s\":", key);
        p = strstr(json, search);
        if (!p) return 0;

        p += strlen(search);
        while (*p == ' ') p++;
        int i = 0;
        while (*p && *p != ',' && *p != '}' && i < out_len - 1)
            out[i++] = *p++;
        while (i > 0 && out[i-1] == ' ') i--;
        out[i] = '\0';
        return i > 0;
    }

    /* 提取带引号的字符串值 */
    p += strlen(search);
    int i = 0;
    while (*p && *p != '"' && i < out_len - 1) {
        if (*p == '\\' && p[1]) {
            p++;
            switch (*p) {
                case 'n': out[i++] = '\n'; break;
                case 't': out[i++] = '\t'; break;
                default:  out[i++] = *p; break;
            }
            p++;
        } else {
            out[i++] = *p++;
        }
    }
    out[i] = '\0';
    return 1;
}

/* 从 JSON 中提取整数字段 */
int json_get_int(const char *json, const char *key) {
    char search[128];
    sprintf(search, "\"%s\":", key);
    const char *p = strstr(json, search);
    if (!p) return 0;
    p += strlen(search);
    while (*p == ' ') p++;
    return atoi(p);
}

/* 从 JSON 中提取浮点数字段 */
double json_get_dbl(const char *json, const char *key) {
    char search[128];
    sprintf(search, "\"%s\":", key);
    const char *p = strstr(json, search);
    if (!p) return 0.0;
    p += strlen(search);
    while (*p == ' ') p++;
    return atof(p);
}
```

```c
/* 获取商品列表（支持类别筛选 + 关键字搜索） */
void handle_get_products(const char *query, char *response_body) {
    char category[STR_LEN] = {0};
    char keyword[STR_LEN] = {0};
    int has_category = 0, has_keyword = 0;

    /* 解析查询参数 */
    if (query && strlen(query) > 0) {
        char qcopy[1024];
        strncpy(qcopy, query, sizeof(qcopy) - 1);
        char *tok = strtok(qcopy, "&");
        while (tok) {
            url_decode(tok);
            if (strncmp(tok, "category=", 9) == 0) {
                strncpy(category, tok + 9, STR_LEN - 1);
                has_category = 1;
            } else if (strncmp(tok, "keyword=", 8) == 0) {
                strncpy(keyword, tok + 8, STR_LEN - 1);
                has_keyword = 1;
            }
            tok = strtok(NULL, "&");
        }
    }

    /* 过滤商品 */
    Product filtered[MAX_PRODUCTS];
    int fcount = 0;
    for (int i = 0; i < product_count; i++) {
        int match = 1;
        if (has_category && strcmp(products[i].category, category) != 0)
            match = 0;
        if (has_keyword && match) {
            if (strstr(products[i].name, keyword) == NULL &&
                strstr(products[i].id, keyword) == NULL &&
                strstr(products[i].manufacturer, keyword) == NULL)
                match = 0;
        }
        if (match) filtered[fcount++] = products[i];
    }

    /* 构建 JSON 响应 */
    char buf[65536];
    JsonBuilder jb;
    json_init(&jb, buf, sizeof(buf));
    json_begin_obj(&jb);
    json_add_int(&jb, "total", fcount);
    json_begin_array(&jb, "data");
    for (int i = 0; i < fcount; i++) {
        /* 逐个序列化为 JSON 对象 ... */
    }
    json_end_array(&jb);
    json_end_obj(&jb);
    strncpy(response_body, buf, 8191);
}

/* 新增商品 */
void handle_create_product(const char *body, char *response_body) {
    if (product_count >= MAX_PRODUCTS) {
        sprintf(response_body, "{\"error\":\"商品数量已达上限\"}");
        return;
    }

    Product *p = &products[product_count];
    next_product_id(p->id);  /* 自动生成编号 */
    json_get_str(body, "name", p->name, STR_LEN);
    json_get_str(body, "category", p->category, STR_LEN);
    json_get_str(body, "manufacturer", p->manufacturer, STR_LEN);
    json_get_str(body, "model", p->model, STR_LEN);
    p->stock = json_get_int(body, "stock");
    p->price = json_get_dbl(body, "price");

    product_count++;
    save_products();
    sprintf(response_body, "{\"message\":\"商品添加成功\",\"id\":\"%s\"}", p->id);
}

/* 删除商品 — 数组元素前移覆盖 */
void handle_delete_product(const char *id, char *response_body) {

    if (idx < 0) {
        sprintf(response_body, "{\"error\":\"未找到商品\"}");
        return;
    }
    for (int i = idx; i < product_count - 1; i++) {
        products[i] = products[i + 1];  /* 结构体整体赋值覆盖 */
    }
    product_count--;
    save_products();
    sprintf(response_body, "{\"message\":\"商品删除成功\"}");
}
```

```c
/* 创建销售记录 — 核心: 自动扣减库存 */
void handle_create_sale(const char *body, char *response_body) {
    if (sale_count >= MAX_SALES) {
        sprintf(response_body, "{\"error\":\"交易记录数量已达上限\"}");
        return;
    }

    /* ① 解析并校验商品编号 */
    char pid[STR_LEN];
    if (!json_get_str(body, "product_id", pid, STR_LEN)) {
        sprintf(response_body, "{\"error\":\"缺少商品编号\"}");
        return;
    }

    /* ② 查找商品是否存在 */
    int pidx = find_product_by_id(pid);
    if (pidx < 0) {
        sprintf(response_body, "{\"error\":\"商品不存在\"}");
        return;
    }

    /* ③ 校验数量 */
    int qty = json_get_int(body, "quantity");
    if (qty <= 0) {
        sprintf(response_body, "{\"error\":\"销售数量必须大于0\"}");
        return;
    }

    /* ④ 校验库存是否充足 */
    if (products[pidx].stock < qty) {
        sprintf(response_body, "{\"error\":\"库存不足，当前库存: %d\"}", products[pidx].stock);
        return;
    }

    /* ⑤ 填充销售记录 */
    Sale *s = &sales[sale_count];
    next_sale_id(s->id);
    strncpy(s->product_id, pid, STR_LEN - 1);
    strncpy(s->name, products[pidx].name, STR_LEN - 1);          /* 冗余复制 */
    strncpy(s->category, products[pidx].category, STR_LEN - 1);
    json_get_str(body, "date", s->date, STR_LEN);
    s->quantity = qty;
    s->price = json_get_dbl(body, "price");
    if (s->price <= 0) s->price = products[pidx].price;          /* 默认用商品单价 */
    s->total = s->quantity * s->price;

    /* ★⑥ 扣减库存 — 关键步骤 ★ */
    products[pidx].stock -= qty;

    /* ⑦ 保存两份文件 */
    sale_count++;
    save_sales();
    save_products();   /* 库存变了也必须保存 */

    /* ⑧ 返回成功信息 */
    JsonBuilder jb;
    json_init(&jb, response_body, 8192);
    json_begin_obj(&jb);
    json_add_str(&jb, "message", "销售记录添加成功");
    json_add_str(&jb, "id", s->id);
    json_add_dbl(&jb, "total", s->total);
    json_add_int(&jb, "remaining_stock", products[pidx].stock);
    json_end_obj(&jb);
}

/* 修改销售记录 — 先恢复旧库存，再扣减新库存 */
void handle_update_sale(const char *id, const char *body, char *response_body) {
    int idx = -1;
    for (int i = 0; i < sale_count; i++)
        if (strcmp(sales[i].id, id) == 0) { idx = i; break; }
    if (idx < 0) {
        sprintf(response_body, "{\"error\":\"未找到交易记录\"}");
        return;
    }

    Sale *s = &sales[idx];
    int qty = json_get_int(body, "quantity");
    if (qty > 0) {
        int pidx = find_product_by_id(s->product_id);
        if (pidx >= 0) {
            products[pidx].stock += s->quantity;   /* ① 恢复旧库存 */
            if (products[pidx].stock < qty) {      /* ② 检查新数量 */
                products[pidx].stock -= s->quantity; /* 回滚 */
                sprintf(response_body, "{\"error\":\"库存不足\"}");
                return;
            }
            products[pidx].stock -= qty;            /* ③ 扣减新库存 */
            save_products();
        }
        s->quantity = qty;
    }
    s->total = s->quantity * s->price;
    save_sales();
    sprintf(response_body, "{\"message\":\"交易记录更新成功\"}");
}

/* 删除销售记录 — 恢复库存(退货) */
void handle_delete_sale(const char *id, char *response_body) {
    int idx = -1;
    for (int i = 0; i < sale_count; i++)
        if (strcmp(sales[i].id, id) == 0) { idx = i; break; }
    if (idx < 0) {
        sprintf(response_body, "{\"error\":\"未找到交易记录\"}");
        return;
    }

    /* ★恢复库存★ */
    int pidx = find_product_by_id(sales[idx].product_id);
    if (pidx >= 0) {
        products[pidx].stock += sales[idx].quantity;
        save_products();
    }

    /* 数组前移删除 */
    for (int i = idx; i < sale_count - 1; i++)
        sales[i] = sales[i + 1];
    sale_count--;
    save_sales();
    sprintf(response_body, "{\"message\":\"交易记录删除成功\"}");
}

/* 销售列表 — 多条件筛选 + qsort 快速排序 */
void handle_get_sales(const char *query, char *response_body) {
    char category[STR_LEN] = {0}, date_from[STR_LEN] = {0};
    char date_to[STR_LEN] = {0}, product_id[STR_LEN] = {0};
    char sort[16] = "desc";

    /* 解析查询参数: category, date_from, date_to, product_id, sort */
    /* ... */

    /* 多条件"与"过滤 */
    Sale filtered[MAX_SALES];
    int fcount = 0;
    for (int i = 0; i < sale_count; i++) {
        int match = 1;
        if (has_category && strcmp(sales[i].category, category) != 0) match = 0;
        if (has_from && strcmp(sales[i].date, date_from) < 0) match = 0;
        if (has_to && strcmp(sales[i].date, date_to) > 0) match = 0;
        if (has_pid && strcmp(sales[i].product_id, product_id) != 0) match = 0;
        if (match) filtered[fcount++] = sales[i];
    }

    /* qsort 快速排序 */
    if (strcmp(sort, "asc") == 0)
        qsort(filtered, fcount, sizeof(Sale), cmp_sale_date_asc);
    else
        qsort(filtered, fcount, sizeof(Sale), cmp_sale_date_desc);

    /* 构建 JSON 响应 ... */
}

/* qsort 比较函数 */
static int cmp_sale_date_desc(const void *a, const void *b) {
    return strcmp(((Sale *)b)->date, ((Sale *)a)->date);
}
```

```c
/* 仪表盘综合统计 — 今日/本周/本月销售 + 库存总览 */
void handle_dashboard_stats(char *response_body) {
    /* 历史总额统计 */
    double total_sales = 0, total_purchases = 0;
    for (int i = 0; i < sale_count; i++)
        total_sales += sales[i].total;
    for (int i = 0; i < purchase_count; i++)
        total_purchases += purchases[i].total;

    /* 获取今日日期 */
    time_t now = time(NULL);
    struct tm *t = localtime(&now);
    char today[16];
    sprintf(today, "%04d-%02d-%02d", t->tm_year + 1900, t->tm_mon + 1, t->tm_mday);

    /* 今日销售额 */
    double today_sales = 0;
    for (int i = 0; i < sale_count; i++)
        if (strcmp(sales[i].date, today) == 0)
            today_sales += sales[i].total;

    /* 本周销售额（回退到周一） */
    int wday = t->tm_wday;
    if (wday == 0) wday = 7;
    struct tm week_start = *t;
    week_start.tm_mday -= (wday - 1);
    mktime(&week_start);
    char week_start_str[16];
    sprintf(week_start_str, "%04d-%02d-%02d",
        week_start.tm_year + 1900, week_start.tm_mon + 1, week_start.tm_mday);

    double week_sales = 0;
    for (int i = 0; i < sale_count; i++)
```

## 九、HTTP 服务器 + 路由分发 (server.c)

```c
#define PORT 8080
#define BUF_SIZE 65536

/* 发送 JSON 格式的 HTTP 响应 (含 CORS 跨域头) */
static void send_response(SOCKET sock, int code, const char *body) {
    char header[512];
    const char *status_text = (code == 200) ? "OK" :
                              (code == 201) ? "Created" :
                              (code == 404) ? "Not Found" : "Internal Server Error";

    int hlen = sprintf(header,
        "HTTP/1.1 %d %s\r\n"
        "Content-Type: application/json; charset=utf-8\r\n"
        "Content-Length: %d\r\n"
        "Access-Control-Allow-Origin: *\r\n"
        "Access-Control-Allow-Methods: GET, POST, PUT, DELETE, OPTIONS\r\n"
        "Access-Control-Allow-Headers: Content-Type\r\n"
        "Connection: close\r\n"
        "\r\n",
        code, status_text, (int)strlen(body));

    send(sock, header, hlen, 0);
    send(sock, body, (int)strlen(body), 0);
}

/* URL 路径匹配: 检查 path 是否以 pattern 开头 */
static const char *path_match(const char *path, const char *pattern) {
    int plen = (int)strlen(pattern);
    if (strncmp(path, pattern, plen) == 0) {
        if (path[plen] == '\0' || path[plen] == '?') return path + plen;
        if (path[plen] == '/') return path + plen + 1;
    }
    return NULL;
}

/* 核心路由函数 — 将 HTTP 请求分发到对应的处理函数 */
static void handle_request(SOCKET sock, const char *method,
                           const char *path, const char *body,
                           int is_local_client) {
    char response[131072];
    const char *rem;

    /* 静态文件服务: / → index.html, /css/*, /js/* */
    if (strcmp(path, "/") == 0) { send_file_response(sock, "frontend/index.html"); return; }
    if ((rem = path_match(path, "/css")) != NULL) { /* 服务 CSS */ return; }
    if ((rem = path_match(path, "/js")) != NULL)  { /* 服务 JS */  return; }

    /* CORS 预检请求 */
    if (strcmp(method, "OPTIONS") == 0) { send_response(sock, 200, ""); return; }

    /* ===== Products API ===== */
    if ((rem = path_match(path, "/api/products/stats")) != NULL) {
        if (strcmp(method, "GET") == 0)
            { handle_product_stats(response); send_response(sock, 200, response); return; }
    }
    if ((rem = path_match(path, "/api/products")) != NULL) {
        if (rem[0] == '\0' || rem[0] == '?') {
            /* 列表: GET(查询) / POST(新增) */
        } else {
            /* 单项: GET / PUT / DELETE */
            char id[STR_LEN];
            get_path_id(rem, id, STR_LEN);
            /* 路由到具体处理函数 */
        }
    }

    /* ===== Sales API ===== */
    if ((rem = path_match(path, "/api/sales")) != NULL) { /* 同理 */ }

    /* ===== Purchases API ===== */
    if ((rem = path_match(path, "/api/purchases")) != NULL) { /* 同理 */ }

    /* ===== Stats API ===== */
    if (strcmp(path, "/api/stats/dashboard") == 0)    { handle_dashboard_stats(response); }
    if (path_match(path, "/api/stats/ranking"))       { handle_sales_ranking(/*...*/); }
    if (strcmp(path, "/api/stats/low-stock") == 0)    { handle_low_stock_alerts(response); }
    if (path_match(path, "/api/stats/weekly"))        { handle_weekly_category_sales(/*...*/); }
    if (strcmp(path, "/api/stats/inventory") == 0)    { handle_inventory_report(response); }

    /* 404 */
    sprintf(response, "{\"error\":\"API endpoint not found: %s %s\"}", method, path);
    send_response(sock, 404, response);
}

/* main() — 服务器启动主函数 */
int main(void) {
    /* ① 加载数据文件 */
    load_products();
    load_sales();
    load_purchases();

    /* ② 初始化 Winsock */
    WSADATA wsa;
    WSAStartup(MAKEWORD(2, 2), &wsa);

    /* ③ 创建 socket + 绑定 + 监听 */
    SOCKET listen_sock = socket(AF_INET, SOCK_STREAM, 0);
    int opt = 1;
    setsockopt(listen_sock, SOL_SOCKET, SO_REUSEADDR, (const char *)&opt, sizeof(opt));

    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;    /* 绑定所有网卡 0.0.0.0 */
    addr.sin_port = htons(PORT);          /* 端口 8080，htons 转网络字节序 */

    bind(listen_sock, (struct sockaddr *)&addr, sizeof(addr));
    listen(listen_sock, 5);

    /* ④ 主循环: accept → 解析 HTTP → 路由 → 响应 */
    while (1) {
        SOCKET client = accept(listen_sock, /*...*/);

        /* 读取 HTTP 请求 */
        char *req_buf = (char *)malloc(BUF_SIZE);
        recv(client, req_buf, /*...*/);

        /* 解析请求行: "GET /api/products?category=笔 HTTP/1.1\r\n" */
        char method[16] = {0}, path[1024] = {0};
        sscanf(req_buf, "%15s %1023s", method, path);

        /* 定位请求体: \r\n\r\n 之后 */
        char *body = strstr(req_buf, "\r\n\r\n");
        if (body) body += 4;

        /* 路由分发 */
        handle_request(client, method, path, body, is_local_client);

        /* 清理 */
        free(req_buf);
        closesocket(client);
    }

    closesocket(listen_sock);
    WSACleanup();
    return 0;
}
```
